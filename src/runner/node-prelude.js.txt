// 워커(V8) 안에서 BOJ node.js 환경을 흉내 내는 최소 shim.
// 호스트가 __stdin(문자열), __write(fd, str)을 넣어 줍니다.
(function () {
    'use strict';
    const g = globalThis;
    const stdinText = g.__stdin;
    const write = g.__write;
    delete g.__stdin;
    delete g.__write;

    const encoder = new TextEncoder();
    const decoder = new TextDecoder();

    // ── Buffer ──────────────────────────────────────────────────────
    // 풀이에서 쓰는 범위만 구현합니다. (입력을 바이트로 읽기, toString, 간단한 변환)
    const TEXT = Symbol('text');
    const view = (b) => new DataView(b.buffer, b.byteOffset, b.byteLength);
    const latin1 = (b) => {
        let s = '';
        for (let i = 0; i < b.length; i += 8192) s += String.fromCharCode.apply(null, b.subarray(i, i + 8192));
        return s;
    };
    class Buffer extends Uint8Array {
        static from(value, encodingOrOffset, length) {
            if (typeof value === 'string') {
                const encoding = String(encodingOrOffset || 'utf8').toLowerCase();
                if (encoding === 'hex') {
                    const out = new Buffer(value.length >> 1);
                    for (let i = 0; i < out.length; i++) out[i] = parseInt(value.substr(i * 2, 2), 16);
                    return out;
                }
                if (encoding === 'base64') {
                    const bin = atob(value);
                    const out = new Buffer(bin.length);
                    for (let i = 0; i < bin.length; i++) out[i] = bin.charCodeAt(i);
                    return out;
                }
                if (encoding === 'latin1' || encoding === 'binary' || encoding === 'ascii') {
                    const out = new Buffer(value.length);
                    for (let i = 0; i < value.length; i++) out[i] = value.charCodeAt(i) & 255;
                    return out;
                }
                const bytes = encoder.encode(value);
                return new Buffer(bytes.buffer, bytes.byteOffset, bytes.byteLength);
            }
            if (value instanceof ArrayBuffer) return new Buffer(value, encodingOrOffset, length);
            const out = new Buffer(value.length >>> 0);
            for (let i = 0; i < out.length; i++) out[i] = value[i];
            return out;
        }
        static alloc(size, fill) {
            const out = new Buffer(size);
            if (fill !== undefined) out.fill(typeof fill === 'string' ? fill.charCodeAt(0) : fill);
            return out;
        }
        static allocUnsafe(size) {
            return new Buffer(size);
        }
        static isBuffer(value) {
            return value instanceof Buffer;
        }
        static byteLength(value) {
            return typeof value === 'string' ? encoder.encode(value).length : value.byteLength;
        }
        static concat(list, totalLength) {
            let total = totalLength;
            if (total === undefined) {
                total = 0;
                for (const b of list) total += b.length;
            }
            const out = new Buffer(total);
            let offset = 0;
            for (const b of list) {
                if (offset >= total) break;
                out.set(b.subarray(0, total - offset), offset);
                offset += b.length;
            }
            return out;
        }
        toString(encoding, start, end) {
            if (encoding === undefined && start === undefined && this[TEXT] !== undefined) return this[TEXT];
            const part = start === undefined && end === undefined ? this : this.subarray(start || 0, end);
            const name = String(encoding || 'utf8').toLowerCase();
            if (name === 'hex') return Array.prototype.map.call(part, (v) => (v < 16 ? '0' : '') + v.toString(16)).join('');
            if (name === 'base64') return btoa(latin1(part));
            if (name === 'latin1' || name === 'binary' || name === 'ascii') return latin1(part);
            return decoder.decode(part);
        }
        toJSON() {
            return { type: 'Buffer', data: Array.from(this) };
        }
        slice(start, end) {
            return this.subarray(start, end);
        }
        equals(other) {
            if (this.length !== other.length) return false;
            for (let i = 0; i < this.length; i++) if (this[i] !== other[i]) return false;
            return true;
        }
        write(text, offset) {
            const bytes = encoder.encode(text);
            const at = offset || 0;
            const n = Math.min(bytes.length, this.length - at);
            this.set(bytes.subarray(0, n), at);
            return n;
        }
        readUInt8(o) { return this[o || 0]; }
        readInt8(o) { return view(this).getInt8(o || 0); }
        readUInt16LE(o) { return view(this).getUint16(o || 0, true); }
        readUInt16BE(o) { return view(this).getUint16(o || 0, false); }
        readInt16LE(o) { return view(this).getInt16(o || 0, true); }
        readInt16BE(o) { return view(this).getInt16(o || 0, false); }
        readUInt32LE(o) { return view(this).getUint32(o || 0, true); }
        readUInt32BE(o) { return view(this).getUint32(o || 0, false); }
        readInt32LE(o) { return view(this).getInt32(o || 0, true); }
        readInt32BE(o) { return view(this).getInt32(o || 0, false); }
        readDoubleLE(o) { return view(this).getFloat64(o || 0, true); }
        readDoubleBE(o) { return view(this).getFloat64(o || 0, false); }
        readBigInt64LE(o) { return view(this).getBigInt64(o || 0, true); }
        readBigUInt64LE(o) { return view(this).getBigUint64(o || 0, true); }
        // Uint8Array와 달리 문자열이나 바이트열도 찾습니다.
        indexOf(value, byteOffset, encoding) {
            if (typeof byteOffset === 'string') {
                encoding = byteOffset;
                byteOffset = 0;
            }
            let start = byteOffset | 0;
            if (start < 0) start = Math.max(0, this.length + start);
            if (typeof value === 'number') return Uint8Array.prototype.indexOf.call(this, value & 255, start);
            const needle = typeof value === 'string' ? Buffer.from(value, encoding) : value;
            if (needle.length === 0) return Math.min(start, this.length);
            outer: for (let i = start; i <= this.length - needle.length; i++) {
                for (let j = 0; j < needle.length; j++) if (this[i + j] !== needle[j]) continue outer;
                return i;
            }
            return -1;
        }
        includes(value, byteOffset, encoding) {
            return this.indexOf(value, byteOffset, encoding) !== -1;
        }
    }
    // writeInt32LE 등: 값을 쓰고 다음 오프셋을 돌려줍니다.
    [
        ['UInt8', 'setUint8', 1], ['Int8', 'setInt8', 1],
        ['UInt16LE', 'setUint16', 2, true], ['UInt16BE', 'setUint16', 2, false],
        ['Int16LE', 'setInt16', 2, true], ['Int16BE', 'setInt16', 2, false],
        ['UInt32LE', 'setUint32', 4, true], ['UInt32BE', 'setUint32', 4, false],
        ['Int32LE', 'setInt32', 4, true], ['Int32BE', 'setInt32', 4, false],
        ['FloatLE', 'setFloat32', 4, true], ['FloatBE', 'setFloat32', 4, false],
        ['DoubleLE', 'setFloat64', 8, true], ['DoubleBE', 'setFloat64', 8, false],
        ['BigInt64LE', 'setBigInt64', 8, true], ['BigUInt64LE', 'setBigUint64', 8, true],
    ].forEach(([name, setter, size, le]) => {
        Object.defineProperty(Buffer.prototype, 'write' + name, {
            writable: true,
            configurable: true,
            value(v, o) {
                const at = o || 0;
                view(this)[setter](at, v, le);
                return at + size;
            },
        });
    });

    let stdinBytes = null;
    const stdinBuffer = () => {
        if (!stdinBytes) stdinBytes = encoder.encode(stdinText);
        const buffer = new Buffer(stdinBytes.buffer, stdinBytes.byteOffset, stdinBytes.byteLength);
        // 가장 흔한 readFileSync(0).toString()은 다시 디코딩하지 않고 원래 문자열을 돌려줍니다.
        buffer[TEXT] = stdinText;
        return buffer;
    };

    // ── util.inspect (Node의 console.log 출력 형식) ─────────────────
    const BREAK_LENGTH = 80;
    const MAX_ARRAY = 100;
    const MAX_DEPTH = 2;
    const IDENTIFIER = /^[A-Za-z_$][A-Za-z0-9_$]*$/;

    const quote = (text) => {
        let q = "'";
        if (text.includes("'")) {
            if (!text.includes('"')) q = '"';
            else if (!text.includes('`') && !text.includes('${')) q = '`';
        }
        const body = text.replace(/[\\\n\r\t\b\f\v]/g, (c) => ({ '\\': '\\\\', '\n': '\\n', '\r': '\\r', '\t': '\\t', '\b': '\\b', '\f': '\\f', '\v': '\\v' })[c]);
        return q + (q === "'" ? body.replace(/'/g, "\\'") : body) + q;
    };

    const className = (value) => {
        let proto = Object.getPrototypeOf(value);
        while (proto) {
            const desc = Object.getOwnPropertyDescriptor(proto, 'constructor');
            if (desc && typeof desc.value === 'function' && desc.value.name) return desc.value.name;
            proto = Object.getPrototypeOf(proto);
        }
        return null;
    };

    const formatError = (error) => {
        const head = error.name + (error.message ? ': ' + error.message : '');
        const frames = String(error.stack || '')
            .split('\n')
            .slice(1)
            .filter((line) => line.includes('main.js'));
        return frames.length ? head + '\n' + frames.join('\n') : head;
    };

    // Node의 groupArrayElements: 긴 배열을 여러 열로 맞춰 줄바꿈합니다.
    const groupArrayElements = (ctx, output, value) => {
        let totalLength = 0;
        let maxLength = 0;
        let outputLength = output.length;
        if (value.length > MAX_ARRAY) outputLength--;
        const dataLen = new Array(outputLength);
        for (let i = 0; i < outputLength; i++) {
            const len = output[i].length;
            dataLen[i] = len;
            totalLength += len + 2;
            if (maxLength < len) maxLength = len;
        }
        const actualMax = maxLength + 2;
        if (actualMax * 3 + ctx.indent < BREAK_LENGTH && (totalLength / actualMax > 5 || maxLength <= 6)) {
            const averageBias = Math.sqrt(actualMax - totalLength / output.length);
            const biasedMax = Math.max(actualMax - 3 - averageBias, 1);
            const columns = Math.min(
                Math.round(Math.sqrt(2.5 * biasedMax * outputLength) / biasedMax),
                Math.floor((BREAK_LENGTH - ctx.indent) / actualMax),
                12,
                15
            );
            if (columns <= 1) return output;
            const tmp = [];
            const maxLineLength = [];
            for (let i = 0; i < columns; i++) {
                let lineLength = 0;
                for (let j = i; j < output.length; j += columns) if (dataLen[j] > lineLength) lineLength = dataLen[j];
                maxLineLength.push(lineLength + 2);
            }
            let padStart = true;
            for (let i = 0; i < output.length; i++) {
                if (typeof value[i] !== 'number' && typeof value[i] !== 'bigint') {
                    padStart = false;
                    break;
                }
            }
            for (let i = 0; i < outputLength; i += columns) {
                const max = Math.min(i + columns, outputLength);
                let str = '';
                let j = i;
                for (; j < max - 1; j++) {
                    const cell = output[j] + ', ';
                    str += padStart ? cell.padStart(maxLineLength[j - i], ' ') : cell.padEnd(maxLineLength[j - i], ' ');
                }
                str += padStart ? output[j].padStart(maxLineLength[j - i] - 2, ' ') : output[j];
                tmp.push(str);
            }
            if (value.length > MAX_ARRAY) tmp.push(output[outputLength]);
            output = tmp;
        }
        return output;
    };

    const reduceToSingleString = (ctx, output, base, braces, isArray, level, value) => {
        const entries = output.length;
        if (isArray && entries > 6) output = groupArrayElements(ctx, output, value);
        if (ctx.deepest - level < 3 && entries === output.length) {
            const start = output.length + ctx.indent + braces[0].length + base.length + 10;
            let total = output.length + start;
            let fits = total + output.length <= BREAK_LENGTH;
            for (let i = 0; fits && i < output.length; i++) {
                total += output[i].length;
                if (total > BREAK_LENGTH) fits = false;
            }
            if (fits && !base.includes('\n')) {
                const joined = output.join(', ');
                if (!joined.includes('\n')) return (base ? base + ' ' : '') + braces[0] + ' ' + joined + ' ' + braces[1];
            }
        }
        const indentation = '\n' + ' '.repeat(ctx.indent);
        return (base ? base + ' ' : '') + braces[0] + indentation + '  ' + output.join(',' + indentation + '  ') + indentation + braces[1];
    };

    const formatValue = (ctx, value, level, top) => {
        if (typeof value === 'string') return top ? value : quote(value);
        if (typeof value === 'number') return Object.is(value, -0) ? '-0' : String(value);
        if (typeof value === 'bigint') return value + 'n';
        if (typeof value === 'symbol') return value.toString();
        if (value === null || typeof value !== 'object' && typeof value !== 'function') return String(value);
        if (ctx.seen.includes(value)) return '[Circular *1]';

        if (typeof value === 'function') {
            const source = Function.prototype.toString.call(value);
            if (/^class\b/.test(source)) {
                const parent = Object.getPrototypeOf(value);
                return '[class ' + (value.name || '(anonymous)') + (parent && parent.name ? ' extends ' + parent.name : '') + ']';
            }
            return value.name ? '[Function: ' + value.name + ']' : '[Function (anonymous)]';
        }
        if (value instanceof Error) return formatError(value);
        if (value instanceof RegExp) return String(value);
        if (value instanceof Date) return isNaN(value.getTime()) ? 'Invalid Date' : value.toISOString();
        if (value instanceof Buffer) {
            const shown = Array.prototype.slice.call(value, 0, 50).map((v) => (v < 16 ? '0' : '') + v.toString(16));
            return '<Buffer' + (shown.length ? ' ' + shown.join(' ') : '') + (value.length > 50 ? ' ... ' + (value.length - 50) + ' more bytes' : '') + '>';
        }
        if (typeof Promise !== 'undefined' && value instanceof Promise) return 'Promise { <pending> }';

        const isArray = Array.isArray(value);
        const isTyped = ArrayBuffer.isView(value) && !(value instanceof DataView);
        const isMap = value instanceof Map;
        const isSet = value instanceof Set;
        const name = className(value);
        const proto = Object.getPrototypeOf(value);

        let prefix = '';
        let braces = ['{', '}'];
        if (isArray) {
            braces = ['[', ']'];
            if (name !== 'Array') prefix = (name || '[Array: null prototype]') + '(' + value.length + ')';
        } else if (isTyped) {
            braces = ['[', ']'];
            prefix = name + '(' + value.length + ')';
        } else if (isMap || isSet) {
            prefix = name + '(' + value.size + ')';
        } else if (proto === null) {
            prefix = '[Object: null prototype]';
        } else if (name && name !== 'Object') {
            prefix = name;
        }

        // 빈 배열/객체는 Node처럼 깊이와 줄바꿈 판단에 넣지 않고 바로 씁니다.
        if ((isArray || (!isTyped && !isMap && !isSet)) && Object.keys(value).length === 0)
            return (prefix ? prefix + ' ' : '') + braces[0] + braces[1];
        if (level > ctx.depth) return isArray ? '[Array]' : '[' + (name || 'Object') + ']';
        if (level > ctx.deepest) ctx.deepest = level;
        ctx.seen.push(value);
        ctx.indent += 2;
        const child = (v) => formatValue(ctx, v, level + 1, false);
        const output = [];
        if (isArray || isTyped) {
            const shown = Math.min(value.length, MAX_ARRAY);
            let holes = 0;
            const flushHoles = () => {
                if (holes) output.push('<' + holes + ' empty item' + (holes > 1 ? 's' : '') + '>');
                holes = 0;
            };
            for (let i = 0; i < shown; i++) {
                if (isArray && !(i in value)) {
                    holes++;
                    continue;
                }
                flushHoles();
                output.push(child(value[i]));
            }
            flushHoles();
            if (value.length > shown) {
                const more = value.length - shown;
                output.push('... ' + more + ' more item' + (more > 1 ? 's' : ''));
            }
        } else if (isMap) {
            value.forEach((v, k) => output.push(child(k) + ' => ' + child(v)));
        } else if (isSet) {
            value.forEach((v) => output.push(child(v)));
        }
        if (!isTyped) {
            for (const key of Object.keys(value)) {
                if ((isArray) && /^(0|[1-9]\d*)$/.test(key)) continue;
                const desc = Object.getOwnPropertyDescriptor(value, key);
                const shownKey = IDENTIFIER.test(key) ? key : quote(key);
                let text;
                if (desc && (desc.get || desc.set)) text = desc.get ? (desc.set ? '[Getter/Setter]' : '[Getter]') : '[Setter]';
                else text = child(value[key]);
                output.push(shownKey + ': ' + text);
            }
        }
        ctx.indent -= 2;
        ctx.seen.pop();

        if (output.length === 0) return (prefix ? prefix + ' ' : '') + braces[0] + braces[1];
        return reduceToSingleString(ctx, output, prefix, braces, isArray || isTyped, level, value);
    };

    const inspect = (value, top, depth = MAX_DEPTH) =>
        formatValue({ seen: [], indent: 0, deepest: 0, depth: depth === null ? Infinity : depth }, value, 0, top);

    const format = (args) => {
        // Node는 인자가 하나뿐이면 형식 문자열을 해석하지 않습니다. (console.log('100%%') → 100%%)
        if (typeof args[0] === 'string' && args.length > 1 && /%[sdifoOjc%]/.test(args[0])) {
            let i = 1;
            const head = args[0].replace(/%([sdifoOjc%])/g, (m, c) => {
                if (c === '%') return '%';
                if (i >= args.length) return m;
                const a = args[i++];
                if (c === 'd') return typeof a === 'bigint' ? a + 'n' : typeof a === 'object' && a !== null ? 'NaN' : inspect(Number(a), true);
                if (c === 'i') return typeof a === 'bigint' ? a + 'n' : String(parseInt(a, 10));
                if (c === 'f') return String(parseFloat(a));
                if (c === 'j') {
                    try {
                        return JSON.stringify(a);
                    } catch (e) {
                        return '[Circular]';
                    }
                }
                if (c === 'c') return '';
                if (c === 's') return typeof a === 'string' ? a : typeof a === 'bigint' ? a + 'n' : typeof a === 'number' ? inspect(a, true) : typeof a === 'object' && a !== null ? inspect(a, false) : String(a);
                return inspect(a, false);
            });
            return [head].concat(args.slice(i).map((a) => inspect(a, true))).join(' ');
        }
        return args.map((a) => inspect(a, true)).join(' ');
    };

    // ── events ──────────────────────────────────────────────────────
    class EventEmitter {
        on(event, fn) {
            const map = this.__listeners || (this.__listeners = {});
            (map[event] = map[event] || []).push(fn);
            return this;
        }
        addListener(event, fn) {
            return this.on(event, fn);
        }
        once(event, fn) {
            const self = this;
            function wrap() {
                self.off(event, wrap);
                fn.apply(self, arguments);
            }
            return this.on(event, wrap);
        }
        off(event, fn) {
            const map = this.__listeners || {};
            map[event] = (map[event] || []).filter((f) => f !== fn);
            return this;
        }
        removeListener(event, fn) {
            return this.off(event, fn);
        }
        removeAllListeners(event) {
            if (event === undefined) this.__listeners = {};
            else if (this.__listeners) delete this.__listeners[event];
            return this;
        }
        emit(event) {
            const args = Array.prototype.slice.call(arguments, 1);
            const list = ((this.__listeners || {})[event] || []).slice();
            list.forEach((fn) => fn.apply(this, args));
            return list.length > 0;
        }
        listenerCount(event) {
            return ((this.__listeners || {})[event] || []).length;
        }
    }
    Object.defineProperty(EventEmitter.prototype, '__listeners', { writable: true, enumerable: false, value: undefined });
    EventEmitter.EventEmitter = EventEmitter;
    // const { once } = require('events'); await once(rl, 'close');
    EventEmitter.once = (emitter, event) => new Promise((resolve) => emitter.once(event, (...args) => resolve(args)));
    const makeEmitter = () => {
        const emitter = new EventEmitter();
        Object.defineProperty(emitter, '__listeners', { writable: true, enumerable: false, value: {} });
        return emitter;
    };

    // ── process ─────────────────────────────────────────────────────
    const stdin = makeEmitter();
    let stdinEncoding = null;
    stdin.fd = 0;
    stdin.isTTY = false;
    stdin.setEncoding = function (encoding) {
        stdinEncoding = encoding || 'utf8';
        return this;
    };
    stdin.resume = stdin.pause = stdin.ref = stdin.unref = function () {
        return this;
    };
    // 'readable' 이벤트 안에서 read()로 읽는 방식: 입력 전체를 한 번 주고 그다음은 null
    let draining = false;
    let stdinTaken = false;
    stdin.read = function () {
        if (!draining || stdinTaken || stdinText.length === 0) return null;
        stdinTaken = true;
        return stdinEncoding ? stdinText : stdinBuffer();
    };

    class ExitSignal {
        constructor(code) {
            this.code = code;
        }
    }
    g.__ExitSignal = ExitSignal;

    const makeStream = (fd) => ({
        fd,
        isTTY: false,
        write(chunk) {
            if (typeof chunk === 'string') write(fd, chunk);
            else if (chunk instanceof Uint8Array) write(fd, decoder.decode(chunk));
            else {
                const received =
                    chunk === null || chunk === undefined
                        ? String(chunk)
                        : typeof chunk === 'object'
                          ? 'an instance of ' + (className(chunk) || 'Object')
                          : 'type ' + typeof chunk + ' (' + inspect(chunk, false) + ')';
                const error = new TypeError(
                    'The "chunk" argument must be of type string or an instance of Buffer, TypedArray, or DataView. Received ' + received
                );
                error.code = 'ERR_INVALID_ARG_TYPE';
                throw error;
            }
            return true;
        },
        end(chunk) {
            if (chunk !== undefined) this.write(chunk);
        },
        on() {
            return this;
        },
        once() {
            return this;
        },
    });

    const startTime = Date.now();
    const process = makeEmitter();
    Object.assign(process, {
        argv: ['node', '/main.js'],
        env: {},
        platform: 'linux',
        version: 'v22.0.0',
        versions: { node: '22.0.0' },
        exitCode: undefined,
        stdin,
        stdout: makeStream(1),
        stderr: makeStream(2),
        exit(code) {
            throw new ExitSignal(code === undefined ? process.exitCode || 0 : code);
        },
        cwd: () => '/',
        uptime: () => (Date.now() - startTime) / 1000,
        hrtime: Object.assign(
            (prev) => {
                const ms = performance.now();
                const t = [Math.floor(ms / 1000), Math.floor((ms % 1000) * 1e6)];
                if (!prev) return t;
                let sec = t[0] - prev[0];
                let nano = t[1] - prev[1];
                if (nano < 0) {
                    sec--;
                    nano += 1e9;
                }
                return [sec, nano];
            },
            { bigint: () => BigInt(Math.floor(performance.now() * 1e6)) }
        ),
        memoryUsage: () => ({ rss: 0, heapTotal: 0, heapUsed: 0, external: 0 }),
        nextTick: (fn, ...args) => Promise.resolve().then(() => fn(...args)),
    });

    // ── fs ──────────────────────────────────────────────────────────
    const STDIN_PATHS = ['/dev/stdin', 0, '0'];
    let readPos = 0;
    const fs = {
        readFileSync(path, options) {
            if (STDIN_PATHS.indexOf(path) === -1) {
                const err = new Error("ENOENT: no such file or directory, open '" + path + "'");
                err.code = 'ENOENT';
                throw err;
            }
            const encoding = typeof options === 'string' ? options : options && options.encoding;
            return encoding ? stdinText : stdinBuffer();
        },
        readSync(fd, buffer, offset, length, position) {
            const all = stdinBuffer();
            const at = typeof offset === 'object' && offset !== null ? offset.offset || 0 : offset || 0;
            const want = typeof offset === 'object' && offset !== null ? offset.length : length;
            const from = typeof position === 'number' ? position : readPos;
            const n = Math.max(0, Math.min(want === undefined ? buffer.length - at : want, all.length - from));
            buffer.set(all.subarray(from, from + n), at);
            if (typeof position !== 'number') readPos += n;
            return n;
        },
        writeSync(fd, data) {
            const text = typeof data === 'string' ? data : data instanceof Uint8Array ? decoder.decode(data) : String(data);
            write(fd === 2 ? 2 : 1, text);
            return text.length;
        },
        writeFileSync(path, data) {
            if (path === '/dev/stdout' || path === 1) write(1, String(data));
            else if (path === '/dev/stderr' || path === 2) write(2, String(data));
        },
        existsSync(path) {
            return STDIN_PATHS.indexOf(path) !== -1;
        },
        openSync(path) {
            return STDIN_PATHS.indexOf(path) !== -1 ? 0 : 3;
        },
        closeSync() {},
        fstatSync: () => ({ size: stdinBuffer().length, isFile: () => true }),
    };

    // ── readline ────────────────────────────────────────────────────
    const interfaces = [];
    const createInterface = (options, outputArg) => {
        const rl = makeEmitter();
        const isOptions = options && typeof options === 'object' && 'input' in options;
        const output = isOptions ? options.output : outputArg;
        let promptText = isOptions && typeof options.prompt === 'string' ? options.prompt : '> ';
        // output을 주면 Node처럼 질문/프롬프트 문자열이 표준 출력에 나갑니다. (BOJ에서는 오답 원인)
        const show = (text) => {
            if (output && typeof output.write === 'function') output.write(String(text));
        };
        Object.defineProperty(rl, '__show', { value: show });
        rl.closed = false;
        rl.close = function () {
            if (!this.closed) {
                this.closed = true;
                this.emit('close');
            }
        };
        rl.question = function (query, cb) {
            show(query);
            this.once('line', cb);
        };
        rl.setPrompt = function (text) {
            promptText = String(text);
        };
        rl.getPrompt = () => promptText;
        rl.prompt = function () {
            show(promptText);
        };
        rl.pause = rl.resume = function () {
            return this;
        };
        let iterator = null;
        rl[Symbol.asyncIterator] = function () {
            // Node처럼 몇 번을 불러도 같은 줄 큐를 공유합니다.
            if (iterator) return iterator;
            const queue = [];
            let done = false;
            let waiting = null;
            rl.on('line', (line) => {
                if (waiting) {
                    const w = waiting;
                    waiting = null;
                    w({ value: line, done: false });
                } else queue.push(line);
            });
            rl.on('close', () => {
                done = true;
                if (waiting) {
                    const w = waiting;
                    waiting = null;
                    w({ value: undefined, done: true });
                }
            });
            iterator = {
                [Symbol.asyncIterator]() {
                    return this;
                },
                next() {
                    if (queue.length) return Promise.resolve({ value: queue.shift(), done: false });
                    if (done) return Promise.resolve({ value: undefined, done: true });
                    return new Promise((resolve) => (waiting = resolve));
                },
                return() {
                    rl.close();
                    return Promise.resolve({ value: undefined, done: true });
                },
            };
            return iterator;
        };
        interfaces.push(rl);
        return rl;
    };
    const readline = { createInterface };
    const readlinePromises = {
        createInterface(options, outputArg) {
            const rl = createInterface(options, outputArg);
            rl.question = function (query) {
                rl.__show(query);
                return new Promise((resolve) => this.once('line', resolve));
            };
            return rl;
        },
    };

    // ── 그 밖의 모듈 ────────────────────────────────────────────────
    const normalize = (path) => {
        const absolute = path.startsWith('/');
        const parts = [];
        for (const part of path.split('/')) {
            if (!part || part === '.') continue;
            if (part === '..') {
                if (parts.length && parts[parts.length - 1] !== '..') parts.pop();
                else if (!absolute) parts.push('..');
            } else parts.push(part);
        }
        return (absolute ? '/' : '') + parts.join('/') || (absolute ? '/' : '.');
    };
    const path = {
        sep: '/',
        delimiter: ':',
        normalize,
        join: (...parts) => normalize(parts.filter(Boolean).join('/')),
        resolve: (...parts) => {
            let out = '';
            for (const part of parts) out = part.startsWith('/') ? part : out + '/' + part;
            return normalize(out.startsWith('/') ? out : '/' + out);
        },
        basename: (p, ext) => {
            const base = p.replace(/\/+$/, '').split('/').pop() || '';
            return ext && base.endsWith(ext) ? base.slice(0, -ext.length) : base;
        },
        dirname: (p) => {
            const i = p.replace(/\/+$/, '').lastIndexOf('/');
            return i < 0 ? '.' : i === 0 ? '/' : p.slice(0, i);
        },
        extname: (p) => {
            const base = p.split('/').pop() || '';
            const i = base.lastIndexOf('.');
            return i <= 0 ? '' : base.slice(i);
        },
        isAbsolute: (p) => p.startsWith('/'),
    };
    path.posix = path;

    const isPrimitive = (v) => v === null || typeof v !== 'object';
    const deepEqual = (a, b, loose) => {
        if (Object.is(a, b)) return true;
        if (isPrimitive(a) || isPrimitive(b)) {
            // assert.deepEqual은 원시값끼리 ==로 비교합니다. (NaN끼리는 같음)
            return !!loose && isPrimitive(a) && isPrimitive(b) && (a == b || (a !== a && b !== b));
        }
        if (Array.isArray(a) !== Array.isArray(b)) return false;
        const ka = Object.keys(a);
        const kb = Object.keys(b);
        return ka.length === kb.length && ka.every((k) => deepEqual(a[k], b[k], loose));
    };
    class AssertionError extends Error {
        constructor(message) {
            super(message);
            this.name = 'AssertionError';
            this.code = 'ERR_ASSERTION';
        }
    }
    const fail = (message, fallback) => {
        throw message instanceof Error ? message : new AssertionError(message || fallback);
    };
    const assert = (value, message) => {
        if (!value) fail(message, 'The expression evaluated to a falsy value');
    };
    Object.assign(assert, {
        AssertionError,
        ok: assert,
        equal: (a, b, m) => a == b || fail(m, inspect(a, false) + ' == ' + inspect(b, false)),
        notEqual: (a, b, m) => a != b || fail(m, inspect(a, false) + ' != ' + inspect(b, false)),
        strictEqual: (a, b, m) => Object.is(a, b) || fail(m, 'Expected values to be strictly equal:\n\n' + inspect(a, false) + ' !== ' + inspect(b, false) + '\n'),
        notStrictEqual: (a, b, m) => !Object.is(a, b) || fail(m, 'Expected values to be strictly unequal'),
        deepEqual: (a, b, m) => deepEqual(a, b, true) || fail(m, 'Expected values to be loosely deep-equal'),
        deepStrictEqual: (a, b, m) => deepEqual(a, b) || fail(m, 'Expected values to be strictly deep-equal'),
        throws: (fn, m) => {
            try {
                fn();
            } catch (e) {
                return;
            }
            fail(typeof m === 'string' ? m : undefined, 'Missing expected exception.');
        },
        fail: (m) => fail(m, 'Failed'),
    });
    assert.strict = assert;

    const util = {
        inspect: (v, options) => inspect(v, false, options && 'depth' in options ? options.depth : MAX_DEPTH),
        format: (...a) => format(a),
        isDeepStrictEqual: (a, b) => deepEqual(a, b),
        promisify:
            (fn) =>
            (...args) =>
                new Promise((resolve, reject) => fn(...args, (err, value) => (err ? reject(err) : resolve(value)))),
        types: {},
        TextEncoder,
        TextDecoder,
    };
    const timers = { setTimeout: (...a) => g.setTimeout(...a), clearTimeout: (...a) => g.clearTimeout(...a), setImmediate: (...a) => g.setImmediate(...a), setInterval: (...a) => g.setInterval(...a), clearInterval: (...a) => g.clearInterval(...a) };
    const modules = {
        fs,
        readline,
        'readline/promises': readlinePromises,
        util,
        process,
        os: { EOL: '\n', platform: () => 'linux', cpus: () => [{}], totalmem: () => 0, freemem: () => 0 },
        path,
        assert,
        'assert/strict': assert,
        events: EventEmitter,
        buffer: { Buffer },
        perf_hooks: { performance },
        timers,
    };

    g.require = (name) => {
        const key = String(name).replace(/^node:/, '');
        if (key in modules) return modules[key];
        const err = new Error("Cannot find module '" + name + "'\n(로컬 실행기는 fs, readline, util, path, assert, events, buffer, os, perf_hooks만 지원합니다.)");
        err.code = 'MODULE_NOT_FOUND';
        throw err;
    };
    g.module = { exports: {} };
    g.global = g;
    g.Buffer = Buffer;
    g.process = process;
    const log = (...a) => write(1, format(a) + '\n');
    const warn = (...a) => write(2, format(a) + '\n');
    const timerLabels = {};
    g.console = {
        log,
        info: log,
        debug: log,
        error: warn,
        warn,
        trace: (...a) => write(2, 'Trace: ' + format(a) + '\n'),
        dir: (v) => write(1, inspect(v, false) + '\n'),
        table: (v) => write(1, inspect(v, false) + '\n'),
        time(label = 'default') {
            timerLabels[label] = performance.now();
        },
        timeEnd(label = 'default') {
            if (label in timerLabels) {
                write(1, label + ': ' + (performance.now() - timerLabels[label]).toFixed(3) + 'ms\n');
                delete timerLabels[label];
            }
        },
        timeLog(label = 'default') {
            if (label in timerLabels) write(1, label + ': ' + (performance.now() - timerLabels[label]).toFixed(3) + 'ms\n');
        },
        group() {},
        groupEnd() {},
        assert(cond, ...a) {
            if (!cond) write(2, 'Assertion failed' + (a.length ? ': ' + format(a) : '') + '\n');
        },
    };
    g.__formatError = formatError;

    // 메인 스크립트가 끝난 뒤 호스트가 호출: 입력 이벤트를 흘려보냅니다.
    g.__drainInput = () => {
        draining = true;
        if (stdin.listenerCount('data') > 0 && stdinText.length > 0) {
            stdinTaken = true;
            stdin.emit('data', stdinEncoding ? stdinText : stdinBuffer());
        }
        if (stdin.listenerCount('readable') > 0) stdin.emit('readable');
        stdin.emit('end');
        stdin.emit('close');
        // readline은 \n, \r\n, 단독 \r을 모두 줄 끝으로 봅니다.
        const lines = stdinText.length ? stdinText.split(/\r\n|\r|\n/) : [];
        if (lines.length && lines[lines.length - 1] === '') lines.pop();
        interfaces.forEach((rl) => {
            for (const line of lines) {
                if (rl.closed) break;
                rl.emit('line', line);
            }
            rl.close();
        });
    };
    g.__exitCode = () => (typeof process.exitCode === 'number' ? process.exitCode : 0);
    // 종료 직전에 호스트가 호출: process.on('exit') 핸들러를 실행합니다.
    g.__emitExit = (code) => {
        process.exitCode = code;
        process.emit('exit', code);
    };
